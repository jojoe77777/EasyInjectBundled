cmake_minimum_required(VERSION 3.20)
project(EasyInjectConfigTests LANGUAGES CXX)
include(CTest)
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/sqlite.cmake")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(easyinject_instance_cfg_tests instance_cfg_test.cpp)
target_compile_features(easyinject_instance_cfg_tests PRIVATE cxx_std_17)
add_test(NAME instance_cfg_sections COMMAND easyinject_instance_cfg_tests)
if(WIN32)
    add_executable(easyinject_launcher_parity_tests launcher_parity_test.cpp)
    target_compile_features(easyinject_launcher_parity_tests PRIVATE cxx_std_17)
    target_compile_definitions(easyinject_launcher_parity_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(easyinject_launcher_parity_tests PRIVATE easyinject_sqlite)
    add_test(NAME launcher_parity COMMAND easyinject_launcher_parity_tests)
    set_tests_properties(launcher_parity PROPERTIES TIMEOUT 30)
    add_executable(easyinject_installer_io_tests installer_io_test.cpp)
    target_compile_features(easyinject_installer_io_tests PRIVATE cxx_std_17)
    add_test(NAME installer_io COMMAND easyinject_installer_io_tests)
    foreach(variant IN ITEMS compatibility lite)
        add_executable(easyinject_dialog_test_${variant} installer_dialog_test.cpp)
        target_compile_features(easyinject_dialog_test_${variant} PRIVATE cxx_std_17)
        target_compile_definitions(easyinject_dialog_test_${variant} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        if(variant STREQUAL "lite")
            target_compile_definitions(easyinject_dialog_test_${variant} PRIVATE TEST_LITE_INSTALLER=1)
        endif()
        target_link_libraries(easyinject_dialog_test_${variant} PRIVATE easyinject_sqlite Advapi32 Bcrypt Ntdll Psapi Shell32 Shlwapi User32 Winhttp Comctl32)
        add_test(NAME installer_dialog_${variant} COMMAND easyinject_dialog_test_${variant})
        set_tests_properties(installer_dialog_${variant} PROPERTIES TIMEOUT 100)
    endforeach()
    foreach(variant IN ITEMS compatibility reduced lite)
        add_executable(easyinject_frontend_test_${variant} native_frontend_test.cpp)
        target_compile_features(easyinject_frontend_test_${variant} PRIVATE cxx_std_17)
        target_compile_definitions(easyinject_frontend_test_${variant} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        if(variant STREQUAL "lite")
            target_compile_definitions(easyinject_frontend_test_${variant} PRIVATE TEST_LITE_INSTALLER=1)
        elseif(variant STREQUAL "reduced")
            target_compile_definitions(easyinject_frontend_test_${variant} PRIVATE EASYINJECT_REDUCED_AV_HEURISTICS=1)
        endif()
        target_link_libraries(easyinject_frontend_test_${variant} PRIVATE easyinject_sqlite Advapi32 Bcrypt Ntdll Psapi Shell32 Shlwapi User32 Winhttp Comctl32)
        add_test(NAME native_frontend_${variant} COMMAND easyinject_frontend_test_${variant})
        set_tests_properties(native_frontend_${variant} PROPERTIES TIMEOUT 70)
    endforeach()
endif()
